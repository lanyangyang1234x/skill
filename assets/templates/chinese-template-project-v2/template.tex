\newcommand{\maybehyperref}[2]{\@ifundefined{r@#1}{#2}{\hyperref[#1]{#2}}}
\newcommand{\exercise@item}{%
  \par\smallskip\refstepcounter{exercise}%
  \edef\exercise@number{\theexercise}%
  \edef\exercise@label{exe:\exercise@number}%
  \edef\exercise@answerlabel{ans:\exercise@number}%
  \expandafter\label\expandafter{\exercise@label}%
  \noindent\maybehyperref{\exercise@answerlabel}{\textbf{\theexercise.}}\quad\ignorespaces}

  \par\medskip\noindent{\headingfont\large 习题}\par\smallskip
  \begingroup\setlength{\parindent}{0pt}\renewcommand{\item}{\exercise@item}%
}{\par\endgroup\medskip}
\newcommand{\answer@item}{%
  \par\smallskip\stepcounter{answeritem}%
  \edef\answer@number{\answer@prefix.\arabic{answeritem}}%
  \edef\answer@label{ans:\answer@number}%
  \edef\answer@exerciselabel{exe:\answer@number}%
  \phantomsection\protected@edef\@currentlabel{\answer@number}%
  \expandafter\label\expandafter{\answer@label}%
  \noindent\maybehyperref{\answer@exerciselabel}{\textbf{\answer@number.}}\quad\ignorespaces}
